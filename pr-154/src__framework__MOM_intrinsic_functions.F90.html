134!> Bit-stable n-th root of x for x in (0, +inf) and integer n >= 1, suitable
135!! for evaluation inside `!$omp target` / `do concurrent` offloaded regions.
136!!
137!! Lowering `x**(1.0/n)` via the compiler produces `exp((1.0/n)*log(x))` — two
138!! transcendentals whose last-bit rounding differs between host libm and CUDA
139!! libdevice. This routine avoids that path entirely: it uses fixed-iteration
140!! Newton on y^n - x = 0, with y^(n-1) evaluated as repeated multiplication,
141!! and one bit-precision-polishing iteration at the end.
142!!
143!! For x in [0, 1] (the case in `MOM_barotropic.F90`'s `bt_rem = av_rem**Instep`)
144!! convergence is rapid because the linear initial guess y0 = 1 - (1-x)/n is
145!! already within a few percent of the true root.
1460elemental function nth_root(x, n) result(root)
147 !$omp declare target
148 real, intent(in) :: x !< Argument, x >= 0 [arbitrary]
149 integer, intent(in) :: n !< Root index, n >= 1
150 real :: root !< x**(1/n) in the same units as x
151
152 integer, parameter :: maxitt = 20 ! Fixed (deterministic) iteration count
153 real :: y, ypow_nm1
154 real :: x_n_r, x_nm1_r
155 integer :: itt, k
156
157 ! Trivial cases — return early to keep loop tight and avoid 0/0 below.
1580 if (n <= 1) then
1590 root = x
1600 return
161 endif
1620 if (x == 0.0) then
1630 root = 0.0
1640 return
165 endif
166
1670 x_n_r = real(n)
1680 x_nm1_r = real(n - 1)
169
170 ! Linear initial guess valid for x in [0, 1] and decent for moderate x > 1.
171 ! For our caller (av_rem in [0, 1], typically near 1), this is within ~1%.
1720 y = 1.0 - (1.0 - x) / x_n_r
173
174 ! Newton iteration: y_{k+1} = ((n-1)*y_k + x / y_k^{n-1}) / n
175 ! All ops are *, +, /. Integer power y^{n-1} is repeated multiplication,
176 ! so there is no `pow`/`exp/log` lowering anywhere in the iteration.
1770 do itt = 1, maxitt
1780 ypow_nm1 = 1.0
1790 do k = 1, n - 1
1800 ypow_nm1 = ypow_nm1 * y
181 enddo
1820 y = (x_nm1_r * y + x / ypow_nm1) / x_n_r
183 enddo

189!> Rescale `a` to the range [0.125, 1) and compute its cube-root exponent.
190!!
191!! This function decomposes `a` into the form `s * x * 2**e` so that `x` is
192!! in the desired range. This is accomplished by computing the integral cube
193!! root of `e` (as a division) and applying the residual to `x`.
1949317128pure subroutine rescale_cbrt(a, x, e_r, s_a)
195 !$omp declare target
196 real, intent(in) :: a
197 !< The number to be rescaled for cube-root computation [A3]
198 real, intent(out) :: x
199 !< The rescaled value of `a` in the range [0.125, 1) [B3]
200 integer(kind=int64), intent(out) :: e_r
201 !< The integral component of the cube-root exponent of `a`.
202 integer(kind=int64), intent(out) :: s_a
203 !< Sign bit of `a`. A nonzero value indicates negative sign.
204
205 integer(kind=int64) :: xb
206 ! Floating point integer representation of `a`
207 integer(kind=int64) :: e_a
208 ! Exponent of `a`
209 integer(kind=int64) :: e_x
210 ! Exponent of `x`
211
212 ! Pack bits of a into xb and extract its exponent and sign.
2139317128 xb = transfer(a, 1_int64)
2149317128 s_a = ibits(xb, signbit, 1)
2159317128 e_a = ibits(xb, expbit, explen) - bias
216
217 ! The floating-point form of `a` with exponent `e` is
218 !
219 ! a = s * (1 + m) * 2**e
220 !
221 ! where (1+m) ∈ [1,2). We want to split 2**e so that (1+m) is rescaled to
222 ! the range [0.125, 1); that is, [2**-3, 2**0).
223 !
224 ! First decompose the exponent `e` into quotient-remainder form:
225 !
226 ! e = 3⌊e/3⌋ + modulo(e,3)
227 !
228 ! Since modulo(e,3) ∈ {0,1,2}, the second term of the following expression is
229 ! in {-3,-2,-1}.
230 !
231 ! e = 3 * (⌊e/3⌋ + 1) + (modulo(e,3) - 3).
232 !
233 ! Here, (modulo(e,3) - 3) is in the range [2**-3, 1) and holds the
234 ! floating-point exponent of `x`.
235 !
236 ! Fortran integer division is round-to-zero. To convert to floor division,
237 ! we use the sign() intrinsic to shift negative values downward.
238 !
239 ! ⌊e/3⌋ = (e + sign(1,e) - 1) / 3
240 !
241 ! ⌊e/3⌋ + 1 reduces to the form below. This is what we call the integral
242 ! cube-root of `a` in the description above.
243
2449317128 e_r = (e_a + sign(1_int64, e_a) + 2) / 3
245
246 ! modulo() is not implemented on all systems, so compute the remainder as
247 ! r = n - 3*q.
248
2499317128 e_x = e_a - e_r * 3
250
251 ! Insert the new 11-bit exponent into xb and write to x and extend the
252 ! bitcount to 12, so that the sign bit is zero and x is always positive.
2539317128 call mvbits(e_x + bias, 0, explen + 1, xb, fraclen)
2549317128 x = transfer(xb, 1.)
2559317128end subroutine rescale_cbrt
