438720 if (CS%Henyey_IGW_background) then
439720 I_x30 = 2.0 / invcosh(CS%N0_2Omega*2.0) ! This is evaluated at 30 deg.
44087120 do i=is,ie

458 endif

461720 if ((.not.CS%physical_OBL_scheme) .and. (CS%Kd /= CS%Kd_tot_ml)) then

4640 I_Hmix = 1.0 / (CS%Hmix + GV%H_subroundoff)
4650 do i=is,ie ; depth(i) = 0.0 ; enddo

4670 depth_c = depth(i) + 0.5*h(i,j,k)

4690 elseif (depth_c >= 2.0*CS%Hmix) then ; Kd_lay(i,k) = Kd_sfc(i)

472 endif
473
4740 depth(i) = depth(i) + h(i,j,k)
475 enddo ; enddo
476
477 else ! There is no vertical structure to the background diffusivity.
4786534720 do k=1,nz ; do i=is,ie
4796534000 Kd_lay(i,k) = Kd_sfc(i)
480 enddo ; enddo
481 endif
482
483 ! Update Kd_int and Kv_bkgnd, based on Kd_lay. These might be just used for diagnostic purposes.
48487120 do i=is,ie
48586400 Kd_int(i,1) = 0.0 ; Kv_bkgnd(i,1) = 0.0
48687120 Kd_int(i,nz+1) = 0.0 ; Kv_bkgnd(i,nz+1) = 0.0
487 enddo
4886447600 do K=2,nz ; do i=is,ie
4896393600 Kd_int(i,K) = 0.5*(Kd_lay(i,k-1) + Kd_lay(i,k))
4906446880 Kv_bkgnd(i,K) = Kd_int(i,K) * CS%prandtl_bkgnd
491 enddo ; enddo
492 endif
