158!> This provides a convenient interface for writing an error message
159!! with run-time filter based on a verbosity and the severity of the error.
1601subroutine MOM_error(level, message, all_print)
161 integer, intent(in) :: level !< The severity level of this message
162 character(len=*), intent(in) :: message !< A message to write out
163 logical, optional, intent(in) :: all_print !< If present and true, any PEs are
164 !! able to write this message.
165 logical :: write_msg
166 integer :: rc
167
1681 if (skip_mpi_dep) then
1690 write_msg = .true.
170 else
1711 write_msg = is_root_pe()
172 endif
1731 if (present(all_print)) write_msg = write_msg .or. all_print
174
1752 select case (level)
176 case (NOTE)
1771 if (write_msg.and.verbosity>=2) call loc_MOM_err(NOTE, message)
178 case (WARNING)
1790 if (write_msg.and.verbosity>=1) call loc_MOM_err(WARNING, message)
180 case (FATAL)
1810 if (ignore_fatal) then
1820 print *, "(FATAL): " // message
1830 rc = kill(err_pid, err_signal)
184 ! NOTE: MPI launchers require, in their words, "a few seconds" to
185 ! propagate the signal to the nodes, so we wait here to avoid
186 ! anomalous FATAL calls.
187 ! In practice, the signal will take control before sleep() completes.
1880 rc = sleep(3)
189 endif
1900 if (verbosity>=0) call loc_MOM_err(FATAL, message)
191 case default
1921 call loc_MOM_err(level, message)
193 end select
