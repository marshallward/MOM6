2850!> Returns change in anomalous pressure change from top to non-dimensional
2851!! position pos between z_t and z_b [R L2 T-2 ~> Pa]
28520real function frac_dp_at_pos(T_t, T_b, S_t, S_b, z_t, z_b, rho_ref, G_e, pos, EOS, frac_dp_bugfix)
2853 real, intent(in) :: T_t !< Potential temperature at the cell top [C ~> degC]
2854 real, intent(in) :: T_b !< Potential temperature at the cell bottom [C ~> degC]
2855 real, intent(in) :: S_t !< Salinity at the cell top [S ~> ppt]
2856 real, intent(in) :: S_b !< Salinity at the cell bottom [S ~> ppt]
2857 real, intent(in) :: z_t !< The geometric height at the top of the layer [Z ~> m]
2858 real, intent(in) :: z_b !< The geometric height at the bottom of the layer [Z ~> m]
2859 real, intent(in) :: rho_ref !< A mean density [R ~> kg m-3], that is subtracted out to
2860 !! reduce the magnitude of each of the integrals.
2861 real, intent(in) :: G_e !< The Earth's gravitational acceleration [L2 Z-1 T-2 ~> m s-2]
2862 real, intent(in) :: pos !< The fractional vertical position, 0 to 1 [nondim]
2863 type(EOS_type), intent(in) :: EOS !< Equation of state structure
2864 logical, intent(in) :: frac_dp_bugfix !< If true, use bugfix in frac_dp_at_pos
2865
2866 ! Local variables
2867 real, parameter :: C1_90 = 1.0/90.0 ! A rational constant [nondim]
2868 real :: dz ! Distance from the layer top [Z ~> m]
2869 real :: top_weight, bottom_weight ! Fractional weights at quadrature points [nondim]
2870 real :: rho_ave ! Average density [R ~> kg m-3]
2871 real, dimension(5) :: T5 ! Temperatures at quadrature points [C ~> degC]
2872 real, dimension(5) :: S5 ! Salinities at quadrature points [S ~> ppt]
2873 real, dimension(5) :: p5 ! Pressures at quadrature points [R L2 T-2 ~> Pa]
2874 real, dimension(5) :: rho5 ! Densities at quadrature points [R ~> kg m-3]
2875 integer :: n
2876
28770 do n=1,5
2878 ! Evaluate density at five quadrature points
28790 bottom_weight = 0.25*real(n-1) * pos
28800 top_weight = 1.0 - bottom_weight
2881 ! Salinity and temperature points are linearly interpolated
28820 S5(n) = top_weight * S_t + bottom_weight * S_b
28830 T5(n) = top_weight * T_t + bottom_weight * T_b
28840 if (frac_dp_bugfix) then
28850 p5(n) = (-1) * ( top_weight * z_t + bottom_weight * z_b ) * ( G_e * rho_ref )
2886 else
28870 p5(n) = ( top_weight * z_t + bottom_weight * z_b ) * ( G_e * rho_ref )
2888 endif !bugfix
2889 enddo
28900 call calculate_density(T5, S5, p5, rho5, EOS)
28910 rho5(:) = rho5(:) !- rho_ref ! Work with anomalies relative to rho_ref
2892
2893 ! Use Boole's rule to estimate the average density
28940 rho_ave = C1_90*(7.0*(rho5(1)+rho5(5)) + 32.0*(rho5(2)+rho5(4)) + 12.0*rho5(3))
2895
28960 dz = ( z_t - z_b ) * pos
28970 frac_dp_at_pos = G_e * dz * rho_ave
28980end function frac_dp_at_pos
