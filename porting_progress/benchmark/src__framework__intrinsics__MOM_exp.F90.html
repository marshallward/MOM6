150 ! 1. Nonfinite handling
151 ! ---------------------
152 ! Nonfinites must be handled first to prevent their appearance in
153 ! calculations, which may raise unwanted floating point signals.
154
155367617 xb = transfer(x, int_mold)
156367617 nonfinite = iand(xb, pos_inf_bits) == pos_inf_bits
157
158367617 if (nonfinite) then
159 ! exp(-Inf) = 0, otherwise pass-through +Inf and +/-NaN values
160 ! Compute x + x to trigger `Invalid` for signaled NaNs.
1610 a = merge(0., x + x, xb == neg_inf_bits)
1620 return
163 endif
164
165 ! 2. Range Reduction
166 ! ------------------
167 ! Apply a range reduction of r = x - K ln2 - (idiv / ndiv) ln2, so that
168 ! exp(x) = 2**K 2**(idiv / ndiv) exp(r).
169 ! If K = nint(x / ln2) then r is in [-ln2/(2*ndiv), ln2/(2*ndiv)] and exp(r)
170 ! can be estimated by a sufficiently accurate polynomial.
171
172 ! Clamp x to [xmin,xmax] to avoid extreme exponents in subnormal handler.
173367617 xc = min(max(x, xmin), xmax)
174
175 ! Compute Z = ndiv K + idiv, where r = x - Z (ln 2 / ndiv)
176367617 Z = NEAREST_INT(xc * n_ln2)
177367617 Zi = transfer(Z + round_bias, int_mold) - round_bias_bits
178
179 ! Compute the subdivision index and integer offset K
180367617 idiv = iand(Zi, idiv_mask)
181367617 K = (Zi - int(idiv, int_kind)) / ndiv
182
183 ! Since Z ~ x N / ln2, the terms in r will nearly cancel and there is some
184 ! expected loss of precision. To compensate, we use a Cody-Waite correction.
185367617 r = (xc - Z * ln2_ndiv_hi) - Z * ln2_ndiv_lo
186
187 ! 3. Polynomial approximation
188 ! ---------------------------
189 ! a = 2**K e where e = 2**(idiv/ndiv) exp(r), and exp(r) = 1 + r * P(r) where
190 ! P(r) is an order-5 Remez minimax polynomial of expm1(r) / r.
191
192367617 idiv_scale = idiv_scale_lookup(idiv)
193367617 idiv_residual = idiv_residual_lookup(idiv)
194367617 expm1_r = exp_remez_expm1_estrin_4(r)
195
196 ! Evaluate the small correction before the final addition to idiv_scale
197367617 e = idiv_scale + idiv_scale * (idiv_residual + r * expm1_r)
198
199 ! 4. Unscaling
200 ! ------------
201 ! Compute a = 2**K e, an exact power-of-2 calculation.
202 ! Adjust scaling to compensate for subnormal output.
203
204 ! exp(r) has range [0.707/ndiv, 1.414/ndiv], so K shifts by either 0 or -1.
205 ! Resolved exponent are in the range {-1022..1023}, so for a to be resolved,
206 ! K must be in {-1021,1023}. (For safety, we actually do {-1020,1022}.)
207
208 ! Determine if K is outside the supported exponent range.
209 ! If so, then apply a bias j to normalize the exponent.
210 ! Kbias is chosen so that the exponent is "something near 1".
211367617 j = merge(Kbias, 0_int_kind, K < Kmin) + merge(-Kbias, 0_int_kind, K > Kmax)
212
213 ! Get the bit representation of e
214367617 eb = transfer(e, int_mold)
215
216 ! Rescale to e to exp(x), possibly including the j bias.
217367617 eb = eb + ishft(k + j, expbit)
218367617 a = transfer(eb, real_mold)
219
220 ! Undo the 2**j bias as floating point multiplication.
221 ! - For "normals", this has no effect.
222 ! - For subnormals, this will force subnormal estimation (if enabled).
223 ! - For resolvable K beyond this range, it triggers an over/underflow.
224 ! - Extreme values of K have already been filtered out by the min/max step.
225367617 fb = ishft(int(expbias, int_kind) - j, expbit)
226367617 a = a * transfer(fb, real_mold)
227367617end procedure exp_repro

263!> An optimized nearest-integer function for floating point reals.
264!!
265!! The value x is shifted from 2**K (1 + a) to 2**(digits-1+K) (1.5 + [a]),
266!! causing the fractional part to be rounded according to the current IEEE
267!! settings. In almost all cases, this is nearest ties-to-even.
268!!
269!! The +0.5 ensures that the biased exponent of negative numbers does not drop
270!! by one, which can cause half-value rounding.
271!!
272!! The behavior of this function does not match nint() or anint(). The nint()
273!! function always ties away from zero, e.g. nint(2.5) = 3.
274!!
275!! It is essential that compilers not reduce (x+b)-b to x. This can typically
276!! be ensured as long as parentheses are respected. This is managed by the
277!! ENABLE_FAST_RINT macro in MOM_exp.h and assigned to NEAREST_INT(). If
278!! unset, then ieee_rint() is used.
279367617pure function fast_rint(x) result(n)
280 real, intent(in) :: x
281 !< Real value to be rounded to the nearest integer
282 real :: n
283 !< Nearest integer to x, stored as a real
284
285367617 n = (x + round_bias) - round_bias
286367617end function fast_rint
