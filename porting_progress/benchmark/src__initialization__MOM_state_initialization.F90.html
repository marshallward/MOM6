1481 type(remapping_CS), pointer :: remap_CS !< Remapping structure for remapping T and S,
1482 !! if associated
1483 real, intent(in) :: z_tol !< The tolerance with which to find the depth
1484 !! matching the specified pressure [Z ~> m].
1485 logical, intent(in) :: frac_dp_bugfix !< If true, use bugfix in frac_dp_at_pos
1486
1487 ! Local variables
14880 real, dimension(nk+1) :: e ! Top and bottom edge positions for reconstructions [Z ~> m]
14890 real, dimension(nk) :: h0, h1 ! Initial and remapped layer thicknesses [H ~> m or kg m-2]
14900 real, dimension(nk) :: S0, S1 ! Initial and remapped layer salinities [S ~> ppt]
14910 real, dimension(nk) :: T0, T1 ! Initial and remapped layer temperatures [C ~> degC]

1493 real :: z_out, e_top ! Interface height positions [Z ~> m]

1496 integer :: k
1497
1498 ! Keep a copy of the initial thicknesses in reverse order to use in remapping
14990 do k=1,nk ; h0(k) = h(nk+1-k) ; enddo
1500
15010 if (GV%Boussinesq) then
15020 min_dz = GV%H_to_Z * min_thickness
1503 ! Calculate original interface positions
15040 e(nk+1) = -depth
15050 do k=nk,1,-1
15060 e(K) = e(K+1) + GV%H_to_Z*h(k)
1507 enddo
1508
15090 P_t = 0.

15150 frac_dp_bugfix=frac_dp_bugfix)

1517 ! Imposed pressure was less that pressure at top of cell
15180 exit
15190 elseif (z_out<=e(K+1)) then
1520 ! Imposed pressure was greater than pressure at bottom of cell
15210 e_top = e(K+1)
1522 else
1523 ! Imposed pressure was fell between pressures at top and bottom of cell
15240 e_top = z_out
15250 exit
1526 endif
15270 P_t = P_b
1528 enddo
15290 if (e_top<e(1)) then

15510 h(k) = min_thickness
1552 else ! This is the last layer that should be removed.
15530 h(k) = h(k) - dh_surf_rem
15540 dh_surf_rem = 0.0
15550 exit
1556 endif
1557 enddo
1558 endif
1559 endif

15640 do k=1,nk
15650 S0(k) = S(nk+1-k)
15660 T0(k) = T(nk+1-k)
15670 h1(k) = h(nk+1-k)
1568 enddo

15700 call remapping_core_h(remap_CS, nk, h0, S0, nk, h1, S1)
15710 do k=1,nk
15720 S(k) = S1(nk+1-k)
15730 T(k) = T1(nk+1-k)
1574 enddo
1575 endif

3190 endif

31970 if (zi_(K) > -1.0*hml) zi_(K) = max(-1.0*hml, Z_bot(i,j))
3198 enddo
